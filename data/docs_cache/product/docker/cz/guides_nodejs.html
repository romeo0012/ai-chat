import bodyParser from 'body-parser';

const app = express();
app.use(bodyParser.json());

app.get('/', (req, res) => {

});

const port = process.env.PORT || 3000;
app.listen(port, () => {

Spusťte testy pomocí dev fáze vašeho Dockerfile:
$ docker compose run --build --rm --no-deps server npm test

Pomocí --no-deps flagu se zabírá startování databáze, protože unit tests nevyžadují ji. Pomocí --rm flagu se odstraní kontejner po dokončení testů. Vy should see the same test output as when running locally.
